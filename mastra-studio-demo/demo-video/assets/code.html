Grounding a Mastra agent in Carver

The retrieval tool is the only thing that changes.

lending-status-agent.ts 
import { Agent } from '@mastra/core/agent';
import { lookupApplicant } from './tools/lookup-applicant';
import { carverLending } from '@carver/sdk';

export const lendingStatusAgent = new Agent({
  id: 'lending-status-agent',
  model: 'openai/gpt-5.6-sol',
  instructions: `${LENDING_STATUS_INSTRUCTIONS}
Search Carver for the obligations that apply, and cite
the source link on anything you rely on.`,
  tools: { lookupApplicant, carverLending },
});
